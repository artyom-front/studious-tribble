    13→let seed = 20260831;
    14→function rng(): number {
    15→  seed |= 0; seed = (seed + 0x6d2b79f5) | 0;
    16→  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    17→  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    18→  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    19→}

   326→        if (cand !== personId) assistPersonId = cand;
   327→      }
   328→      events.push({ minute: uniqueMinute(), type: isPen ? "PENALTY" : "GOAL", personId, teamId: m.home, assistPersonId });
   329→    }
   330→    for (let g = 0; g < awayGoals; g++) {
   331→      const personId = scorerFrom(awayLineup, awaySquad);
   332→      const isPen = rng() < 0.12;
   333→      let assistPersonId: string | undefined;
   334→      if (!isPen && rng() < 0.6) {
   335→        const cand = pick(awayLineup).id;
   336→        if (cand !== personId) assistPersonId = cand;
   337→      }
   338→      events.push({ minute: uniqueMinute(), type: isPen ? "PENALTY" : "GOAL", personId, teamId: m.away, assistPersonId });
   339→    }
   340→
   341→    const yc = ri(0, 3);
   342→    for (let y = 0; y < yc; y++) {
   343→      const home = rng() < 0.5;
   344→      const lineup = home ? homeLineup : awayLineup;
   345→      const p = pick(lineup);
   346→      if (events.some((e) => e.personId === p.id && e.type === "RED_CARD")) continue;
   347→      events.push({ minute: uniqueMinute(), type: "YELLOW_CARD", personId: p.id, teamId: home ? m.home : m.away });
   348→    }
   349→
   350→    if (forced?.redCard) {
   351→      const lineup = forced.redCard.side === "home" ? homeLineup : awayLineup;
   352→      const p = pick(lineup.filter((x) => x.position !== "GK"));
   353→      events.push({ minute: forced.redCard.minute, type: "RED_CARD", personId: p.id, teamId: forced.redCard.side === "home" ? m.home : m.away });
   354→    }
   355→    if (forced?.yellows) {
   356→      for (const y of forced.yellows) {
   357→        const lineup = y.side === "home" ? homeLineup : awayLineup;
   358→        const p = y.personId
   359→          ? lineup.find((x) => x.id === y.personId)
   360→          : pick(lineup.filter((x) => x.position === "DF"));
   361→        if (!p) continue;
   362→        if (events.some((e) => e.personId === p.id && (e.type === "RED_CARD" || e.type === "YELLOW_CARD"))) continue;
   363→        events.push({ minute: uniqueMinute(), type: "YELLOW_CARD", personId: p.id, teamId: y.side === "home" ? m.home : m.away });
   364→      }
   365→    }
   366→
   367→    events.sort((a, b) => a.minute - b.minute);
   368→    for (const e of events) {
   369→      await db.matchEvent.create({ data: { ...e, matchId: m.id } });
   370→    }
   371→
   372→    await completeMatch(m.id, null);
