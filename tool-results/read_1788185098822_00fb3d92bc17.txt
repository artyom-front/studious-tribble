    13→let seed = 20260831;
    14→function rng(): number {
    15→  seed |= 0; seed = (seed + 0x6d2b79f5) | 0;
    16→  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    17→  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    18→  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    19→}

   327→      events.push({ minute: uniqueMinute(), type: isPen ? "PENALTY" : "GOAL", personId, teamId: m.home, assistPersonId });
   328→    }
   329→    for (let g = 0; g < awayGoals; g++) {
   330→      const personId = scorerFrom(awayLineup, awaySquad);
   331→      const isPen = rng() < 0.12;
   332→      let assistPersonId: string | undefined;
   333→      if (!isPen && rng() < 0.6) {
   334→        const cand = pick(awayLineup).id;
   335→        if (cand !== personId) assistPersonId = cand;
   336→      }
   337→      events.push({ minute: uniqueMinute(), type: isPen ? "PENALTY" : "GOAL", personId, teamId: m.away, assistPersonId });
   338→    }
   339→
   340→    const yc = ri(0, 3);
   341→    for (let y = 0; y < yc; y++) {
   342→      const home = rng() < 0.5;
   343→      const lineup = home ? homeLineup : awayLineup;
   344→      const p = pick(lineup);
   345→      if (events.some((e) => e.personId === p.id && e.type === "RED_CARD")) continue;
   346→      events.push({ minute: uniqueMinute(), type: "YELLOW_CARD", personId: p.id, teamId: home ? m.home : m.away });
   347→    }
   348→
   349→    if (forced?.redCard) {
   350→      const lineup = forced.redCard.side === "home" ? homeLineup : awayLineup;
   351→      const p = pick(lineup.filter((x) => x.position !== "GK"));
   352→      events.push({ minute: forced.redCard.minute, type: "RED_CARD", personId: p.id, teamId: forced.redCard.side === "home" ? m.home : m.away });
   353→    }
   354→    if (forced?.yellows) {
   355→      for (const y of forced.yellows) {
   356→        const lineup = y.side === "home" ? homeLineup : awayLineup;
   357→        const p = y.personId
   358→          ? lineup.find((x) => x.id === y.personId)
   359→          : pick(lineup.filter((x) => x.position === "DF"));
   360→        if (!p) continue;
   361→        if (events.some((e) => e.personId === p.id && (e.type === "RED_CARD" || e.type === "YELLOW_CARD"))) continue;
   362→        events.push({ minute: uniqueMinute(), type: "YELLOW_CARD", personId: p.id, teamId: y.side === "home" ? m.home : m.away });
   363→      }
   364→    }
   365→
   366→    events.sort((a, b) => a.minute - b.minute);
   367→    for (const e of events) {
   368→      await db.matchEvent.create({ data: { ...e, matchId: m.id } });
   369→    }
   370→
   371→    await completeMatch(m.id, null);
   372→  };
