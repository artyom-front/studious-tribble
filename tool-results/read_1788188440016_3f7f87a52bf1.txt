    61→/** LIVE-часы: идущая минута + время начала (обновляется каждые 15 секунд) */
    62→function LiveClock({ kickoff }: { kickoff: string }) {
    63→  const [, setTick] = useState(0);
    64→  useEffect(() => {
    65→    const t = setInterval(() => setTick((x) => x + 1), 15000);
    66→    return () => clearInterval(t);
    67→  }, []);
    68→  const start = new Date(kickoff);
    69→  const startStr = start.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit", timeZone: "Europe/Moscow" });
    70→  const elapsed = Math.floor((Date.now() - start.getTime()) / 60000);
    71→  const minuteLabel = elapsed >= 95 ? "90+" : `${Math.max(0, Math.min(90, elapsed))}'`;
    72→  return (
    73→    <span className="flex items-center justify-center gap-1.5 text-[11px] font-bold text-live">
    74→      <span className="h-1.5 w-1.5 rounded-full bg-live live-dot" />
    75→      матч идёт · {minuteLabel} · с {startStr}
    76→    </span>
    77→  );
    78→}

   345→function TimelineTab({ m }: { m: NonNullable<MatchDetail["match"]> }) {
   346→  if (m.events.length === 0) {
   347→    return (
   348→      <div className="p-3">
   349→        <EmptyState
   350→          icon={<BallIcon className="h-6 w-6 opacity-50 text-ink3" />}
   351→          title="Событий пока нет"
   352→          hint={m.status === "SCHEDULED" ? "Матч ещё не начался — события появятся во время игры" : "Протокол пуст"}
   353→        />
   354→      </div>
   355→    );
   356→  }
   357→
   358→  // объединяем SUB_OUT + SUB_IN одной минуты и команды в одну строку «Замена»
   359→  type Row =
   360→    | { kind: "event"; e: EventRow }
   361→    | { kind: "sub"; minute: number; teamId: string; out: EventRow; inEvent: EventRow | null };
   362→  const rows: Row[] = [];
   363→  const consumed = new Set<string>();
   364→  for (const e of m.events) {
   365→    if (consumed.has(e.id)) continue;
   366→    if (e.type === "SUB_OUT") {
   367→      const partner = m.events.find(
   368→        (x) => x.id !== e.id && !consumed.has(x.id) && x.type === "SUB_IN" && x.teamId === e.teamId && Math.abs(x.minute - e.minute) <= 1
   369→      );
   370→      if (partner) {
   371→        consumed.add(partner.id);
   372→        rows.push({ kind: "sub", minute: e.minute, teamId: e.teamId, out: e, inEvent: partner });
   373→        continue;
   374→      }
   375→    }
   376→    rows.push({ kind: "event", e });
   377→  }
   378→
   379→  // текущий счёт после каждого гола (бегущий счёт в хронологии)
   380→  let runHome = 0, runAway = 0;
   381→  const runningScore = (e: EventRow): { home: number; away: number } | null => {
   382→    if (e.type !== "GOAL" && e.type !== "PENALTY" && e.type !== "OWN_GOAL") return null;
   383→    const forHome = e.teamId === m.homeTeam.id;
   384→    const own = e.type === "OWN_GOAL";
   385→    if ((forHome && !own) || (!forHome && own)) runHome++;
   386→    else runAway++;
   387→    return { home: runHome, away: runAway };
   388→  };
   389→
   390→  return (
   391→    <div className="p-3">
   392→      <div className="space-y-1">
   393→        {rows.map((row) => {
   394→          if (row.kind === "sub") {
   395→            const isHome = row.teamId === m.homeTeam.id;
   396→            return (
   397→              <div
   398→                key={row.out.id}
   399→                className={cn(
   400→                  "flex w-full items-center gap-3 rounded-xl border border-sline/50 bg-s2/30 px-3 py-2 text-sm",
   401→                  !isHome && "flex-row-reverse text-right"
   402→                )}
   403→              >
   404→                <span className="flex w-11 shrink-0 items-center justify-center rounded-md bg-s2 py-0.5 font-mono text-xs text-ink2">
   405→                  {row.minute}&apos;
   406→                </span>
   407→                <span className="flex shrink-0 items-center justify-center"><EventIcon type="SUB_OUT" /></span>
   408→                <span className="min-w-0 flex-1">

   567→// ================= Личные встречи (H2H) =================

   569→function H2HTab({ m, h2h }: { m: NonNullable<MatchDetail["match"]>; h2h: MatchDetail["h2h"] }) {
   570→  if (h2h.list.length === 0) {
   571→    return <div className="p-3"><EmptyState title="Команды ещё не встречались" hint="Это их первая встреча" /></div>;
   572→  }
   573→  const { summary } = h2h;
   574→  return (
   575→    <div className="p-3">
   576→      {/* Сводка противостояния */}
   577→      <div className="mb-3 grid grid-cols-3 gap-2">
   578→        <div className="rounded-xl bg-ok/10 px-3 py-2 text-center">
   579→          <p className="font-mono text-xl font-bold text-ok">{summary.homeWins}</p>
   580→          <p className="truncate text-[10px] font-medium uppercase tracking-wide text-ink3">победы · {m.homeTeam.name}</p>
   581→        </div>
   582→        <div className="rounded-xl bg-s2 px-3 py-2 text-center">
   583→          <p className="font-mono text-xl font-bold text-ink">{summary.draws}</p>
   584→          <p className="text-[10px] font-medium uppercase tracking-wide text-ink3">ничьи</p>
   585→        </div>
   586→        <div className="rounded-xl bg-live/10 px-3 py-2 text-center">
   587→          <p className="font-mono text-xl font-bold text-live">{summary.awayWins}</p>
   588→          <p className="truncate text-[10px] font-medium uppercase tracking-wide text-ink3">победы · {m.awayTeam.name}</p>
   589→        </div>
   590→      </div>
   591→
   592→      <div className="space-y-1">
   593→        {h2h.list.map((g) => {
   594→          const homeIsOurHome = g.homeTeam.id === m.homeTeam.id;
   595→          const score = matchScore(g);
   596→          // результат с точки зрения текущих хозяев
   597→          const ourGoals = score ? (homeIsOurHome ? score.home : score.away) : null;
   598→          const theirGoals = score ? (homeIsOurHome ? score.away : score.home) : null;
   599→          const res = score ? (ourGoals! > theirGoals! ? "W" : ourGoals! < theirGoals! ? "L" : "D") : null;
   600→          return (
   601→            <button
   602→              key={g.id}
   603→              onClick={() => navigate(`/match/${g.id}`)}
   604→              className="flex w-full items-center gap-3 rounded-xl border border-sline/50 bg-s2/30 px-3 py-2.5 text-left text-sm transition-colors hover:border-gold/40"
   605→            >
   606→              <span
   607→                className={cn(
   608→                  "flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[10px] font-bold",
   609→                  res === "W" ? "bg-ok text-white" : res === "L" ? "bg-live text-white" : "bg-ink3 text-s0"
   610→                )}
   611→                title={res === "W" ? `Победа ${m.homeTeam.name}` : res === "L" ? `Победа ${m.awayTeam.name}` : "Ничья"}
   612→              >
   613→                {res ?? "·"}
   614→              </span>
   615→              <span className="w-14 shrink-0 text-[11px] text-ink3">
   616→                {new Date(g.kickoff).toLocaleDateString("ru-RU", { day: "2-digit", month: "2-digit", year: "2-digit", timeZone: "Europe/Moscow" })}
   617→              </span>
   618→              <span className="min-w-0 flex-1 truncate text-ink2">{g.homeTeam.name} — {g.awayTeam.name}</span>
   619→              <span className="hidden truncate text-[10px] text-ink3 md:block">{g.season.league}</span>
   620→              <span className={cn("shrink-0 font-mono text-sm font-bold", g.status === "WALKOVER" ? "text-amber-400" : "text-ink")}>
   621→                {score ? `${score.home}:${score.away}` : "WO"}
   622→              </span>
   623→            </button>
   624→          );
   625→        })}
   626→      </div>
   627→    </div>
   628→  );
